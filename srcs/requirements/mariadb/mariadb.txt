DOCKERFILE:
# Dockerfile MariaDB (database for WordPress)
# Build context: srcs/requirements/mariadb

# Base image
FROM debian:bookworm

# Install MariaDB server
RUN apt-get update && apt-get install -y mariadb-server && rm -rf /var/lib/apt/lists/*

# Create the MySQL data directory
# Set the ownership of the MySQL data directory to the mysql user and group
# -R flag is used to apply the ownership change recursively to all files and directories within /var/run/mysqld
RUN mkdir -p /var/run/mysqld && chown -R mysql:mysql /var/run/mysqld

# Copy the custom MariaDB server configuration file to the appropriate location
COPY conf/50-server.cnf /etc/mysql/mariadb.conf.d/50-server.cnf

# Entrypoint script. Copied into /usr/local/bin because that is the
# conventional location for locally-installed executables and is on PATH.
COPY tools/initialization.sh /usr/local/bin/initialization.sh

RUN chmod +x /usr/local/bin/initialization.sh

# it records that this image listens on 3306 for anyone reading `docker inspect`.
# Reachability between containers comes from the shared `inception` network, not from this line.
EXPOSE 3306

# Start the MariaDB server using the entrypoint script - is PID 1 (receives signals)
ENTRYPOINT ["/usr/local/bin/initialization.sh"]

50-SERVER:
# 50-server.cnf — read only by the MariaDB server, not by clients.

[mysqld]

# Drop privileges to the unprivileged mysql user after startup.
user            = mysql

# Mount point of the MariaDB volume.
# Everything written here survives docker compose down and a VM reboot.
datadir         = /var/lib/mysql

# socket path to connect with mariadb client
socket          = /run/mysqld/mysqld.sock

# set's the path to the pid files. Without it, every docker up will ccreate a new one
pid-file        = /run/mysqld/mysqld.pid

port            = 3306

# Listen on every interface in this container's network.
bind-address    = 0.0.0.0

# Compare hosts as raw IPs only. (Using the user+host records takes more time)
skip-name-resolve

# Full 4-byte UTF-8, so WordPress content can hold emoji and non-Latin text.
character-set-server  = utf8mb4
collation-server      = utf8mb4_unicode_ci

INITIALIZATION.SH: (draft)
#!/bin/bash

service mysql start;

mysql -e "CREATE DATABASE IF NOT EXISTS \`${MDB_DB}\`;"

mysql -e "CREATE USER IF NOT EXISTS \`${MDB_USR}\`@'localhost' IDENTIFIED BY '${MDB_PSSWRD}';"

mysql -e "GRANT ALL PRIVILEGES ON \`${MDB_DB}\`.* TO \`${MDB_USR}\`@'%' IDENTIFIED BY '${MDB_PSSWRD}';"

# do I need that?
mysql -e "ALTER USER 'root'@'localhost' IDENTIFIED BY '${MDB_RPSSWRD}';"

mysql -e "FLUSH PRIVLEGES;"

mysqladmin -u -p $MDB_RPSSWRD shutdown

exec mysqld_safe
