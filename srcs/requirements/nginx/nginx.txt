# nginx Dockerfile
# Build context: srcs/requirements/nginx

FROM debian:bookworm

# install  nginx and openssl
RUN apt-get update && apt-get install -y nginx openssl \
	&& rm -rf /var/lib/apt/lists/*

# create a folder for key and certificate
# generate the SSL(TSL) certificate (since is local, we can generate key and certificate together)
# - req -x509: generate a self-signed certificate
# - nodes: do not encrypt the private key
# - days 365: set the validity of the certificate to 365 days
# - newkey rsa:2048: generate a new RSA key with 2048 bits
# -out: specify the output file for the certificate
# -keyout: specify the output file for the private key
# - subj: set the subject of the certificate (C=country, ST=state, L=locality, O=organization, OU=organizational unit, CN=common name, UID=user id)
RUN mkdir -p /etc/nginx/ssl && \
    openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
    -out /etc/nginx/ssl/nginxincep.crt \
    -keyout /etc/nginx/ssl/nginxincep.key \
    -subj "/C=FR/ST=IDF/L=Paris/O=42/OU=42/CN=afontele.42.fr/UID=afontele" \
    -addext "subjectAltName=DNS:afontele.42.fr"
# copy config to the container (copy from build context to the container filesystem root)
COPY conf/nginx.conf /etc/nginx/nginx.conf

EXPOSE 443

# setting nginx to run in the foreground
CMD [ "nginx", "-g", "daemon off;" ]

NGINX.CONF:
# Configuration file for NGINX (Server)

# Master process starts as root (needed to bind 443, a privileged port),
# then forks workers that drop to this unprivileged user.
# Deliberately omits `include sites-enabled/*` and `include conf.d/*`:
# that is what prevents Debian's default port-80 site from ever loading.

# Master process starts as root (needed to bind 443, a privileged port),
# then forks workers that drop to this unprivileged user.
user www-data;

# One worker per CPU core, decided at startup.
worker_processes auto;

pid /run/nginx.pid;

events {
    # Max simultaneous connections a single worker will hold open.
    worker_connections 768;
	# how many connections one worker may hold open. With 4 cores: 4 × 768 ≈ 3072 total.
}

http {
    # Maps file extensions to Content-Type headers. Without it every asset
    # is served as octet-stream and the browser refuses to apply the CSS,
    # so WordPress renders as unstyled text.
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    access_log /var/log/nginx/access.log;
    error_log  /var/log/nginx/error.log;

    server {
        listen 443 ssl;
        server_name afontele.42.fr;

        # Self-signed pair generated at image build time.
        ssl_certificate     /etc/nginx/ssl/nginxincep.crt;
        ssl_certificate_key /etc/nginx/ssl/nginxincep.key;

        ssl_protocols TLSv1.2 TLSv1.3;

		# wordpress volume
        root  /var/www/html;
        index index.php;

        location / {
            try_files $uri $uri/ /index.php?$args;
        }

        location ~ \.php$ {
            include       fastcgi_params;

            # Absolute path of the script, as php-fpm's filesystem sees it.
            fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;

            fastcgi_pass  wordpress:9000;
        }
    }
}
